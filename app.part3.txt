  function contractReadiness() {
    const checks = [
      { label: '参照が3件以上', done: state.references.length >= 3 },
      { label: '要素を3種類以上分解', done: DIMENSIONS.filter(d => state.decomposition[d.id]?.refId).length >= 3 },
      { label: '目的と対象を設定', done: !!state.remix.projectGoal.trim() && !!state.remix.targetAudience.trim() },
      { label: '方向性ワードを3つ設定', done: state.remix.directionWords.filter(x => x.trim()).length === 3 },
      { label: 'Signatureと避けるものを設定', done: !!state.remix.signature.trim() && !!state.remix.avoidList.trim() }
    ];
    return checks;
  }

  function renderOutputs() {
    const contract = generateContract();
    $('#contractOutput').textContent = contract;
    const checks = contractReadiness();
    const score = Math.round(checks.filter(x => x.done).length / checks.length * 100);
    $('#contractScore').textContent = `${score}%`;
    $('#readinessList').innerHTML = checks.map(x => `<div class="ready-item ${x.done ? 'done' : ''}"><span class="ready-dot"></span>${esc(x.label)}</div>`).join('');
    $('#promptOutput').textContent = generatePrompt();
  }

  async function exportProject() {
    const payload = { ...clone(state), exportedAt: new Date().toISOString(), images: {} };
    for (const ref of state.references) {
      if (ref.hasImage) {
        const image = await dbGet(ref.id).catch(() => null);
        if (image) payload.images[ref.id] = image;
      }
    }
    const name = (state.remix.projectName || 'visual-remix-project').trim().toLowerCase().replace(/[^a-z0-9\u3040-\u30ff\u3400-\u9fff]+/g, '-').replace(/^-|-$/g, '') || 'visual-remix-project';
    downloadText(`${name}.visual-remix.json`, JSON.stringify(payload, null, 2), 'application/json');
    showToast('プロジェクトを書き出しました');
  }

  async function importProject(file) {
    try {
      const parsed = JSON.parse(await file.text());
      if (!parsed || !Array.isArray(parsed.references)) throw new Error('Invalid project');
      await dbClear();
      if (parsed.images) {
        for (const [id, data] of Object.entries(parsed.images)) await dbSet(id, data);
      }
      delete parsed.images; delete parsed.exportedAt;
      state = {
        ...clone(DEFAULT_STATE), ...parsed,
        remix: { ...clone(DEFAULT_STATE.remix), ...(parsed.remix || {}) },
        builder: { ...clone(DEFAULT_STATE.builder), ...(parsed.builder || {}) }
      };
      saveState();
      syncAllFields();
      await renderReferences(); renderDecomposition(); renderPalette(); renderOutputs();
      showToast('プロジェクトを読み込みました');
    } catch (e) {
      console.error(e); showToast('読み込みに失敗しました');
    }
  }

  }
